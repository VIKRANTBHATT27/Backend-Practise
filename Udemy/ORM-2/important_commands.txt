S-1: pnpm init
S-2: pnpm add -D typescript tsx @types/node
S-3: pnpm exec tsc --init
S-4: create src/server.ts
S-5: config package.json

    "scripts": {
    "dev": "tsx watch src/server.ts",
    "build": "tsc",
    "start": "node dist/server.js"
    }

S-6: pnpm dev or pnpm run dev
